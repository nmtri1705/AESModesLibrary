cmake_minimum_required(VERSION 3.16)
project(AESModesLab VERSION 3.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(AES_LIBRARY_TYPE "BOTH" CACHE STRING "STATIC, SHARED or BOTH")
set_property(CACHE AES_LIBRARY_TYPE PROPERTY STRINGS STATIC SHARED BOTH)
option(AES_BUILD_EXECUTABLE "Build vector tests" ON)
option(AES_BUILD_TESTS "Register vector tests" ON)
string(TOUPPER "${AES_LIBRARY_TYPE}" AES_LIBRARY_TYPE)
if(NOT AES_LIBRARY_TYPE MATCHES "^(STATIC|SHARED|BOTH)$")
 message(FATAL_ERROR "Invalid AES_LIBRARY_TYPE")
endif()
if(AES_BUILD_TESTS AND NOT AES_BUILD_EXECUTABLE)
 message(FATAL_ERROR "Tests require AES_BUILD_EXECUTABLE")
endif()
foreach(kind STATIC SHARED)
 if(AES_LIBRARY_TYPE STREQUAL kind OR AES_LIBRARY_TYPE STREQUAL "BOTH")
  string(TOLOWER ${kind} suffix)
  set(t aes_modes_${suffix})
  add_library(${t} ${kind} src/library.cpp)
  target_compile_features(${t} PUBLIC cxx_std_17)
  set_target_properties(${t} PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME ${t})
  target_include_directories(${t} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  if(kind STREQUAL "SHARED")
   target_compile_definitions(${t} PUBLIC AES_MODES_SHARED PRIVATE AES_MODES_BUILD)
  endif()
  if(AES_BUILD_EXECUTABLE)
   add_executable(aes_tests_${suffix} tests/main.cpp)
   target_link_libraries(aes_tests_${suffix} PRIVATE ${t})
   if(AES_BUILD_TESTS)
    add_test(NAME vectors_${suffix} COMMAND aes_tests_${suffix} ${CMAKE_CURRENT_SOURCE_DIR}/tests/modes_vectors.txt)
   endif()
  endif()
  install(TARGETS ${t} EXPORT AESModesTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
 endif()
endforeach()
install(FILES include/aes_modes.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT AESModesTargets NAMESPACE AESModes:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AESModes)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/AESModesConfig.cmake "include(\"\${CMAKE_CURRENT_LIST_DIR}/AESModesTargets.cmake\")\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/AESModesConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AESModes)
